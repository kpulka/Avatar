Create `src/core/dispose.ts`:

```ts
import * as THREE from 'three';

/** Material slots that hold shared, scene-level resources and must never be disposed per avatar. */
const SHARED_TEXTURE_KEYS = new Set(['envMap']);

/**
 * Releases GPU resources of a detached subtree: geometries, materials, material textures, skeletons.
 * Each resource is disposed at most once (dedup via Set). Scene-level IBL (envMap) is skipped.
 */
export function disposeObject3D(root: THREE.Object3D): void {
  const geometries = new Set<THREE.BufferGeometry>();
  const materials = new Set<THREE.Material>();
  const textures = new Set<THREE.Texture>();
  const skeletons = new Set<THREE.Skeleton>();

  root.traverse((obj) => {
    const mesh = obj as THREE.Mesh;
    if (mesh.geometry) geometries.add(mesh.geometry);
    const mats = Array.isArray(mesh.material) ? mesh.material : mesh.material ? [mesh.material] : [];
    for (const m of mats) materials.add(m);
    const skinned = obj as THREE.SkinnedMesh;
    if (skinned.isSkinnedMesh && skinned.skeleton) skeletons.add(skinned.skeleton);
  });

  for (const m of materials) {
    for (const [key, value] of Object.entries(m)) {
      if (value instanceof THREE.Texture && !SHARED_TEXTURE_KEYS.has(key)) textures.add(value);
    }
  }

  textures.forEach((t) => t.dispose());
  materials.forEach((m) => m.dispose());
  geometries.forEach((g) => g.dispose());
  skeletons.forEach((s) => s.dispose());
}
```
