1. Add module-level shared uniforms and an updater (preallocate temporaries — runs every frame):
```ts
import { Color, DirectionalLight, Camera, MeshPhysicalMaterial, Vector3 } from 'three';

/** Shared by every skin material; updated once per frame (see StudioLighting.addToScene). */
export const skinLightUniforms = {
  uKeyLightDirView: { value: new Vector3(0, 0, 1) }, // direction TO the light, view space (matches three's directLight.direction)
  uKeyLightColor: { value: new Color(1, 1, 1) },     // color × intensity
  uSkinSSSEnabled: { value: 1 },
};
const _p = new Vector3(); const _t = new Vector3();
export function updateSkinLightUniforms(key: DirectionalLight, camera: Camera): void {
  key.getWorldPosition(_p); key.target.getWorldPosition(_t);
  skinLightUniforms.uKeyLightDirView.value.subVectors(_p, _t).normalize().transformDirection(camera.matrixWorldInverse);
  skinLightUniforms.uKeyLightColor.value.copy(key.color).multiplyScalar(key.intensity);
}
export function setSkinSSSEnabled(on: boolean): void { skinLightUniforms.uSkinSSSEnabled.value = on ? 1 : 0; }
```
2. In `onBeforeCompile` assign the SAME objects (not copies) so updates propagate:
   `shader.uniforms.uKeyLightDirView = skinLightUniforms.uKeyLightDirView;` (same for color/enabled), declare
   `uniform vec3 uKeyLightDirView; uniform vec3 uKeyLightColor; uniform float uSkinSSSEnabled;` in the `<common>`
   replacement, and replace the injected block after `#include <lights_fragment_begin>` with:
```glsl
      // Wrap-diffuse SSS from the KEY light. NOTE: `directLight` here would be the LAST light of the loop (rim).
      float NdotLk = dot(normal, uKeyLightDirView);
      float wrapDiffuse = clamp((NdotLk + uSkinWrap) / (1.0 + uSkinWrap), 0.0, 1.0);
      float scatterEdge = smoothstep(-uSkinWrap, uSkinWrap, NdotLk) * (1.0 - smoothstep(0.0, 0.5, NdotLk));
      float wrapGain = max(wrapDiffuse - clamp(NdotLk, 0.0, 1.0), 0.0); // light that wraps past the terminator
      vec3 sss = material.diffuseColor * uSkinSSSColor * uKeyLightColor * uSkinSSSIntensity * (scatterEdge + wrapGain);
      reflectedLight.directDiffuse += sss * uSkinSSSEnabled;
```
   (`normal` = perturbed normal incl. normal map; `material.diffuseColor` = albedo in the physical pipeline.)
   Set `material.customProgramCacheKey = () => 'skin-sss-v2';`.
