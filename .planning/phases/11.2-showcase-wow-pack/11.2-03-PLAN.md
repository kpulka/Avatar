```js
function wavHeader(dataLen, rate) {
  // 44-byte RIFF/WAVE header, PCM16 mono (src/audio/pcm-to-wav.ts is TypeScript and cannot be imported by a plain .mjs script)
  const h = Buffer.alloc(44);
  h.write('RIFF', 0, 'ascii');
  h.writeUInt32LE(36 + dataLen, 4);
  h.write('WAVE', 8, 'ascii');
  h.write('fmt ', 12, 'ascii');
  h.writeUInt32LE(16, 16);        // fmt chunk size
  h.writeUInt16LE(1, 20);         // audioFormat = PCM
  h.writeUInt16LE(1, 22);         // channels = 1
  h.writeUInt32LE(rate, 24);      // sampleRate
  h.writeUInt32LE(rate * 2, 28);  // byteRate = rate * channels * 2
  h.writeUInt16LE(2, 32);         // blockAlign
  h.writeUInt16LE(16, 34);        // bitsPerSample
  h.write('data', 36, 'ascii');
  h.writeUInt32LE(dataLen, 40);
  return h;
}
```

```ts
import type { CharAlignment } from '../types/avatar';

export interface DemoClip { text: string; sampleRate: number; pcm: Uint8Array; alignment: CharAlignment; }

/** Parses a PCM16 mono WAV without resampling (decodeAudioData would resample to the context rate). */
export function parseWavPcm16(buf: ArrayBuffer): { sampleRate: number; pcm: Uint8Array } {
  const v = new DataView(buf);
  const tag = (o: number) => String.fromCharCode(v.getUint8(o), v.getUint8(o + 1), v.getUint8(o + 2), v.getUint8(o + 3));
  if (tag(0) !== 'RIFF' || tag(8) !== 'WAVE') throw new Error('Not a WAV file');
  let offset = 12; let sampleRate = 0; let bits = 0; let channels = 0;
  while (offset + 8 <= v.byteLength) {
    const id = tag(offset); const size = v.getUint32(offset + 4, true); const body = offset + 8;
    if (id === 'fmt ') { channels = v.getUint16(body + 2, true); sampleRate = v.getUint32(body + 4, true); bits = v.getUint16(body + 14, true); }
    if (id === 'data') {
      if (bits !== 16 || channels !== 1) throw new Error(`Unsupported WAV: ${bits}-bit, ${channels} ch`);
      return { sampleRate, pcm: new Uint8Array(buf, body, Math.min(size, v.byteLength - body)) };
    }
    offset = body + size + (size % 2);
  }
  throw new Error('WAV has no data chunk');
}

const cache = new Map<string, Promise<DemoClip>>();
/** Loads `<base>.wav` + `<base>.json` once (BASE_URL-aware for GitHub Pages). */
export function loadDemoClip(name = 'demo-hello'): Promise<DemoClip> {
  const base = `${import.meta.env.BASE_URL}demo/${name}`;
  if (!cache.has(base)) {
    cache.set(base, (async () => {
      const [wavRes, metaRes] = await Promise.all([fetch(`${base}.wav`), fetch(`${base}.json`)]);
      if (!wavRes.ok || !metaRes.ok) throw new Error('Demo clip unavailable');
      const { sampleRate, pcm } = parseWavPcm16(await wavRes.arrayBuffer());
      const meta = await metaRes.json() as { text: string; alignment: CharAlignment };
      return { text: meta.text, sampleRate, pcm, alignment: meta.alignment };
    })().catch((e) => { cache.delete(base); throw e; }));
  }
  return cache.get(base)!;
}
```
