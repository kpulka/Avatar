```js
// Usage: npm run dev (port 5310) in another terminal, then: node scripts/capture-hero.mjs
import { chromium } from 'playwright';
const browser = await chromium.launch({ args: ['--autoplay-policy=no-user-gesture-required'] });
const ctx = await browser.newContext({ viewport: { width: 1280, height: 720 }, recordVideo: { dir: 'docs/media/raw', size: { width: 1280, height: 720 } } });
const page = await ctx.newPage();
await page.goto('http://localhost:5310/');
await page.waitForSelector('html[data-avatar-ready="1"]', { timeout: 60_000 });
await page.waitForSelector('button[aria-label="Listen to the avatar"]:not([disabled])', { timeout: 10_000 });
await page.waitForTimeout(1500);                       // idle blink/gaze before speaking
await page.getByRole('button', { name: 'Listen to the avatar' }).click();
await page.waitForTimeout(14_000);
await ctx.close(); await browser.close();
console.log('Raw webm in docs/media/raw — convert with ffmpeg (see plan).');
```

`ffmpeg -i docs/media/raw/<file>.webm -vf "fps=15,scale=800:-1:flags=lanczos,split[a][b];[a]palettegen[p];[b][p]paletteuse" docs/media/hero.gif`.
Add `docs/media/raw/` to `.gitignore`.
